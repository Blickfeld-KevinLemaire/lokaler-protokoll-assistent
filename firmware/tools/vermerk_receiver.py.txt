from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
import time

UPLOAD_DIR = Path(__file__).with_name("uploads")
UPLOAD_DIR.mkdir(exist_ok=True)


def unique_target(name: str) -> Path:
    """Nie eine vorhandene Aufnahme ueberschreiben. Mehrere Geraete zaehlen ihre Dateien unabhaengig
    (H5_001, H5_002 ...) und koennen dieselbe Nummer erreichen: dann wird _2, _3 ... angehaengt."""
    target = UPLOAD_DIR / name
    n = 2
    while target.exists():
        target = UPLOAD_DIR / f"{Path(name).stem}_{n}{Path(name).suffix}"
        n += 1
    return target


class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/vermerk-status.txt":
            body = b"VERMERK_SERVER_OK"
            self.send_response(200)
            self.send_header("Content-Type", "text/plain")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
        else:
            self.send_error(404)

    def do_POST(self):
        if self.path != "/upload":
            self.send_error(404)
            return

        self.connection.settimeout(10)
        start = time.monotonic()
        filename = Path(self.headers.get("X-Filename", "upload.wav")).name
        length = int(self.headers.get("Content-Length", "0"))
        target = unique_target(filename)
        temp_target = UPLOAD_DIR / (target.name + ".part")

        remaining = length
        with open(temp_target, "wb") as f:
            while remaining > 0:
                chunk = self.rfile.read(min(65536, remaining))
                if not chunk:
                    break
                f.write(chunk)
                remaining -= len(chunk)

        if remaining == 0:
            temp_target.replace(target)
            body = b"OK"
            self.send_response(200)
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
            duration = max(time.monotonic() - start, 0.001)
            print(f"Gespeichert: {target.name} ({length} Bytes) in {duration:.1f} s "
                  f"= {length / 1024 / duration:.0f} KB/s")
        else:
            temp_target.unlink(missing_ok=True)
            self.send_error(400, "Upload unvollstaendig")

print("VERMERK Testserver auf Port 8765")
ThreadingHTTPServer(("0.0.0.0", 8765), Handler).serve_forever()
